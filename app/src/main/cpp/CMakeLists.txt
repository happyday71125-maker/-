cmake_minimum_required(VERSION 3.22)
project(prollama_jni C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(llama.cpp build_llama EXCLUDE_FROM_ALL)
add_library(prollama_jni SHARED prollama_jni.cpp)
target_include_directories(prollama_jni PRIVATE llama.cpp/include llama.cpp/ggml/include)
find_library(LOG_LIB log)
target_link_libraries(prollama_jni PRIVATE llama ggml ${LOG_LIB})
target_link_options(prollama_jni PRIVATE -Wl,--exclude-libs,ALL -Wl,--gc-sections)
